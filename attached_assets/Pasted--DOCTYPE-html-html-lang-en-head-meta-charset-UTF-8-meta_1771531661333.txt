function cycleStatus(li) {
  let current = 0;
  STATES.forEach((s, i) => { if (s && li.classList.contains(s)) current = i; });
  const next = (current + 1) % STATES.length;
  STATES.forEach(s => { if (s) li.classList.remove(s); });

  li.querySelector('.check-box').textContent = ICONS[next];
}
