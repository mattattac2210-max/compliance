function expandCust(ce){
  if(!ce.recurring||ce.recurring==='none')return[{...ce,date:new Date(ce.date)}];
  const out=[],end=new Date(curY+2,11,31);let cur=new Date(ce.date);
  while(cur<=end){
    out.push({...ce,id:ce.id+'-'+cur.toISOString().slice(0,7),date:new Date(cur)});
    if(ce.recurring==='monthly')cur.setMonth(cur.getMonth()+1);
    else if(ce.recurring==='quarterly')cur.setMonth(cur.getMonth()+3);
    else if(ce.recurring==='annual')cur.setFullYear(cur.getFullYear()+1);
    else break;
  }
  return out;
}

  cells.forEach(c=>{

    const isTd=cd.getTime()===today.getTime(),isSel=selDate&&selDate.getTime()===cd.getTime();
    const evts=c.o?[]:forD(c.y,c.m,c.d);
    const hasOD=evts.some(e=>e.daysUntil<0&&e.type!=='ops');
    let cls='cc2';if(c.o)cls+=' om';if(isTd)cls+=' td';if(isSel)cls+=' sl';if(hasOD)cls+=' ovr';
    const vis=evts.slice(0,3),ov=evts.length-3;
    const chips=vis.map(ev=>{
      const col=ev.isCustom?(ev.customColor||'rgba(255,255,255,.35)'):tc(ev.type,ev.gate),fi=filed.has(ev.id);
      return`<div class="ec" style="background:${col}18;border-left-color:${col};color:${col};opacity:${fi?.5:1}" onclick="event.stopPropagation();selD(${c.y},${c.m},${c.d})">${fi?'✓ ':ev.icon+' '}${ev.short}</div>`;
    }).join('');
    const ovEl=ov>0?`<div class="ovc" onclick="event.stopPropagation();selD(${c.y},${c.m},${c.d})">+${ov} more</div>`:'';
    const div=document.createElement('div');div.className=cls;div.innerHTML=`<div class="dn">${c.d}</div>${chips}${ovEl}`;div.onclick=()=>selD(c.y,c.m,c.d);con.appendChild(div);
  });
}

function renderDet(){
  const panel=document.getElementById('detpanel');
  if(!selDate){panel.innerHTML=`<div class="de2"><div class="ico">📅</div><p>Click any day to see compliance events</p></div>`;return;}
  const evts=forD(selDate.getFullYear(),selDate.getMonth(),selDate.getDate()),ds=fmtD(selDate);
  if(!evts.length){panel.innerHTML=`<div class="dph"><div><div class="dpd">${ds}</div><div class="dpc">No events</div></div><div class="dpx" onclick="closeD()">×</div></div><div class="de2" style="padding:20px"><p>No filings or deadlines on this date.</p></div>`;return;}
  const cards=evts.map(ev=>{
    const col=ev.isCustom?(ev.customColor||'rgba(255,255,255,.35)'):tc(ev.type,ev.gate);
    const gn=ev.gate>=0?(GN[ev.gate]||'Custom'):'Custom';
    const fi=filed.has(ev.id),od=ev.daysUntil<0&&ev.type!=='ops'&&ev.type!=='banjar';
    const ovb=od?`<span style="font-size:9px;color:var(--red);font-family:Montserrat;font-weight:700;margin-left:6px">⚠ OVERDUE</span>`:ev.daysUntil===0?`<span style="font-size:9px;color:var(--gold);font-family:Montserrat;font-weight:700;margin-left:6px">DUE TODAY</span>`:'';
    const rb=ev.recurring?`<span class="rb">↻ Recurring</span>`:'';
    const fb=ev.recurring?(fi?`<button class="db fi">✓ Filed</button>`:`<button class="db pr" onclick="markF('${ev.id}')">Mark as filed</button>`):'';
    const eb=ev.isCustom?`<button class="db de" onclick="delC('${ev.id}')">Delete</button><button class="db se" onclick="editC('${ev.id}')">Edit</button>`:'';
    const vb=!ev.isCustom&&ev.type!=='ops'?`<button class="db se">Go to Vault →</button>`:'';
    return`<div class="dc${ev.isCustom?' cx':''}" style="border-left-color:${col};${od?'background:rgba(239,68,68,.05)':''}">
      <div class="dct"><div style="flex:1"><div class="dcti">${ev.icon} ${ev.title}${ovb}</div></div><div class="dcg" style="background:${col}22;color:${col};border:1px solid ${col}44">${gn}</div></div>
      <div class="dcp">${ev.period||''} ${rb}</div>
      <div class="dcd">${ev.desc||''}</div>
      <div class="dca">${fb}${vb}${eb}</div>
    </div>`;
  }).join('');
  panel.innerHTML=`<div class="dph"><div><div class="dpd">${ds}</div><div class="dpc">${evts.length} event${evts.length!==1?'s':''}</div></div><div class="dpx" onclick="closeD()">×</div></div><div class="dpe">${cards}</div>`;
}

}
