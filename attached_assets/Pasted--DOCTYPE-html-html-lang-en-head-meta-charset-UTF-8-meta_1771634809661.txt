const STORE = {
  events: [],           // all generated + custom events
  filed: new Set(),     // event IDs marked as filed
  snoozed: new Set(),   // event IDs snoozed from alerts
  dismissed: new Set(), // event IDs dismissed from alerts
  custom: [],           // user-created events (persisted)

  // Subscribe to changes
  _subs: [],
  on(fn){ this._subs.push(fn); },
  emit(){ this._subs.forEach(f=>f()); },

  markFiled(id){ this.filed.add(id); this.emit(); },
  unmarkFiled(id){ this.filed.delete(id); this.emit(); },
  snooze(id){ this.snoozed.add(id); this.emit(); },
  dismiss(id){ this.dismissed.add(id); this.emit(); },

  // Get all active events for a given filter
  getFiltered(filter){
    const all = [...this.events, ...this.custom.flatMap(expandCust)];
    if(filter.has('all')) return all;
    return all.filter(ev=>filter.has(ev.type)||(filter.has('custom')&&ev.isCustom));
  },
  // Get alerts (overdue + due within 14 days, not dismissed/snoozed)
  getAlerts(){
    const all = [...this.events, ...this.custom.flatMap(expandCust)];
    return all.filter(ev=>{
      if(this.dismissed.has(ev.id)||this.snoozed.has(ev.id)||this.filed.has(ev.id)) return false;
      if(ev.type==='ops') return false; // holidays not alerted
      return ev.daysUntil<=14;
    }).sort((a,b)=>a.daysUntil-b.daysUntil);
  }
};

function expandCust(ce){
  if(!ce.recurring||ce.recurring==='none') return[{...ce,date:new Date(ce.date)}];
  const out=[],end=new Date(curY+2,11,31);
  let cur=new Date(ce.date);
  while(cur<=end){
    out.push({...ce,id:ce.id+'-'+cur.toISOString().slice(0,7),date:new Date(cur),daysUntil:dU(cur)});
    if(ce.recurring==='monthly') cur.setMonth(cur.getMonth()+1);
    else if(ce.recurring==='quarterly') cur.setMonth(cur.getMonth()+3);
    else if(ce.recurring==='annual') cur.setFullYear(cur.getFullYear()+1);
    else break;
  }
  return out;
}

function renderList(){
  const lv=document.getElementById('listView'); lv.innerHTML='';
  // Get all events in current month (or nearby) sorted by date
  let evts=[];
  for(let mo=curM;mo<Math.min(curM+3,12);mo++){
    evts.push(...forM(curY,mo));
  }
  evts.sort((a,b)=>a.date-b.date);
  // Group by date
  const groups={};
  evts.forEach(ev=>{const k=ev.date.toISOString().slice(0,10);if(!groups[k])groups[k]=[];groups[k].push(ev);});
  if(!Object.keys(groups).length){lv.innerHTML=`<div class="de2"><p>No events in this period.</p></div>`;return;}
  Object.entries(groups).forEach(([dateKey,evs])=>{
    const date=new Date(dateKey+'T00:00:00');
    const today=new Date(); today.setHours(0,0,0,0);
    const diff=Math.floor((date-today)/86400000);
    const relStr=diff===0?'Today':diff===1?'Tomorrow':diff===-1?'Yesterday':diff<0?`${Math.abs(diff)}d ago`:`In ${diff}d`;
    const items=evs.map(ev=>{
      const col=ev.isCustom?(ev.customColor||'#94A3B8'):tc(ev.type,ev.gate);
      const fi=STORE.filed.has(ev.id);
      const od=diff<0&&ev.type!=='ops'&&ev.type!=='banjar';
      const statusBadge=od?`<span class="overdue-badge">${SVG.warn} Overdue</span>`:diff===0?`<span class="due-today-badge">Due today</span>`:diff<=7&&diff>0?`<span class="due-soon-badge">In ${diff}d</span>`:'';
      const recBadge=ev.recurring?`<span class="rb">${SVG.rec} Recurring</span>`:'';
      const gateBadge=ev.gate>=0?`<span class="lev-tag" style="background:${col}14;color:${col}">${GN[ev.gate]||'G'+ev.gate}</span>`:'';
      const fiBtn=ev.recurring?(fi?`<button class="lev-btn filed" disabled>${SVG.check} Filed</button>`:`<button class="lev-btn" onclick="STORE.markFiled('${ev.id}');renderAll()">File</button>`):'';
      const editBtn=ev.isCustom?`<button class="lev-btn" onclick="editC('${ev.id}')">${SVG.edit}</button><button class="lev-btn" onclick="delC('${ev.id}')">${SVG.trash}</button>`:'';
      return`<div class="list-ev" style="opacity:${fi?.6:1}" onclick="openEventDetail('${ev.id}')">
        <div class="lev-color" style="background:${col}"></div>
        <div class="lev-icon">${ev.icon}</div>
        <div class="lev-body">
          <div class="lev-title">${ev.title}${statusBadge}</div>
          <div class="lev-sub">${gateBadge}${recBadge}<span style="font-family:'IBM Plex Mono',monospace;font-size:9px;color:var(--t3)">${ev.period||''}</span></div>
        </div>
        <div class="lev-actions">${fiBtn}${editBtn}</div>
      </div>`;
    }).join('');
    const g=document.createElement('div'); g.className='list-group';
    g.innerHTML=`<div class="lg-head"><div class="lg-date">${fmtD(date)}</div><div class="lg-sub">${relStr} · ${evs.length} event${evs.length!==1?'s':''}</div></div>${items}`;
    lv.appendChild(g);
  });
}

function renderAlerts(){
  const alerts=STORE.getAlerts();
  const overdue=alerts.filter(a=>a.daysUntil<0);
  const today=alerts.filter(a=>a.daysUntil===0);
  const soon=alerts.filter(a=>a.daysUntil>0&&a.daysUntil<=7);
  const upcoming=alerts.filter(a=>a.daysUntil>7&&a.daysUntil<=14);
  const total=overdue.length+today.length;

  // Update badge counts
  document.getElementById('alertCount').textContent=alerts.length;
  document.getElementById('sbAlertCount').textContent=alerts.length;
  document.getElementById('alertSubtitle').textContent=`${overdue.length} overdue · ${today.length} due today · ${soon.length} this week`;
  document.getElementById('alertBtn').classList.toggle('active',total>0);

  function alertItems(items){
    return items.map(ev=>{
      const col=ev.isCustom?(ev.customColor||'#94A3B8'):tc(ev.type,ev.gate);
      const diff=ev.daysUntil;
      const dateStr=diff<0?`${Math.abs(diff)}d overdue`:diff===0?'Due today':`In ${diff} day${diff!==1?'s':''}`;
      const priorityBadge=ev.priority==='critical'?`<span style="font-size:9px;font-weight:700;color:var(--red);background:var(--rt);padding:1px 5px;border-radius:5px;margin-left:5px">Critical</span>`:ev.priority==='high'?`<span style="font-size:9px;font-weight:700;color:var(--gold);background:rgba(217,119,6,.08);padding:1px 5px;border-radius:5px;margin-left:5px">High</span>`:'';
      return`<div class="al-item" style="border-left-color:${col}">
        <div class="al-item-head">
          <div style="flex:1">
            <div class="al-item-title">${ev.title}${priorityBadge}</div>
            <div class="al-item-date">${dateStr} · ${fmtDs(ev.date)}</div>
          </div>
        </div>
        <div class="al-item-desc">${ev.desc||''}</div>
        <div class="al-actions">
          <button class="al-btn done" onclick="STORE.markFiled('${ev.id}');renderAll()">${SVG.check} Mark done</button>
          <button class="al-btn snz" onclick="STORE.snooze('${ev.id}');renderAll()">Snooze 7d</button>
          <button class="al-btn snz" onclick="STORE.dismiss('${ev.id}');renderAll()">Dismiss</button>

  }

  let html='';
  if(!alerts.length){html=`<div class="al-empty">No active alerts. All compliance items are on track.</div>`;}
  else{
    if(overdue.length) html+=`<div class="al-section"><div class="al-sec-title" style="color:var(--red)">Overdue (${overdue.length})</div>${alertItems(overdue)}</div>`;
    if(today.length)   html+=`<div class="al-section"><div class="al-sec-title" style="color:var(--gold)">Due Today (${today.length})</div>${alertItems(today)}</div>`;
    if(soon.length)    html+=`<div class="al-section"><div class="al-sec-title" style="color:var(--blue)">This Week (${soon.length})</div>${alertItems(soon)}</div>`;
    if(upcoming.length)html+=`<div class="al-section"><div class="al-sec-title">Next 2 Weeks (${upcoming.length})</div>${alertItems(upcoming)}</div>`;
  }
  document.getElementById('alertsBody').innerHTML=html;
}

let dragState={evId:null,title:'',color:'#E8192C',active:false};

function startDrag(e,evId,title,color){
  dragState={evId,title,color,active:true};
  e.dataTransfer?.setData('evid',evId);
  const ghost=document.getElementById('dragGhost');
  ghost.textContent=title.slice(0,24);
  ghost.style.background=color;
  ghost.style.display='block';
  moveDragGhost(e);
}

function moveDragGhost(e){
  const g=document.getElementById('dragGhost');
  g.style.left=(e.clientX+12)+'px';
  g.style.top=(e.clientY-10)+'px';
}

document.addEventListener('mousemove',e=>{
  if(dragState.active) moveDragGhost(e);
});
document.addEventListener('mouseup',()=>{
  if(dragState.active){
    dragState.active=false;
    document.getElementById('dragGhost').style.display='none';
    document.querySelectorAll('.drag-over').forEach(el=>el.classList.remove('drag-over'));
  }
});

function dropEvent(evId,y,m,d){
  document.getElementById('dragGhost').style.display='none';
  dragState.active=false;
  // Find the event in custom events and update its date
  const idx=STORE.custom.findIndex(e=>e.id===evId);
  if(idx>=0){
    STORE.custom[idx].date=new Date(y,m,d);
    STORE.custom[idx].daysUntil=dU(new Date(y,m,d));
    renderAll();
    showToast(`Event moved to ${fmtDs(new Date(y,m,d))}`);
  } else {
    // Non-custom: offer to create override
    showToast('System events cannot be dragged — add a custom override instead');
  }
}

// Touch drag support
let touchDrag={active:false,evId:null,startX:0,startY:0};
document.addEventListener('touchstart',e=>{
  const el=e.target.closest('[data-evid]');
  if(el){touchDrag={active:true,evId:el.dataset.evid,startX:e.touches[0].clientX,startY:e.touches[0].clientY};}
},{ passive:true });
document.addEventListener('touchmove',e=>{
  if(!touchDrag.active)return;
  const ghost=document.getElementById('dragGhost');
  ghost.style.display='block';
  ghost.style.left=(e.touches[0].clientX+10)+'px';
  ghost.style.top=(e.touches[0].clientY-20)+'px';
  document.querySelectorAll('.drag-over').forEach(el=>el.classList.remove('drag-over'));
  const el=document.elementFromPoint(e.touches[0].clientX,e.touches[0].clientY);
  el?.closest('.cc2,.wc')?.classList.add('drag-over');
},{ passive:true });
document.addEventListener('touchend',e=>{
  if(!touchDrag.active)return;
  document.getElementById('dragGhost').style.display='none';
  const el=document.elementFromPoint(e.changedTouches[0].clientX,e.changedTouches[0].clientY);
  const target=el?.closest('.cc2,.wc');
  if(target){
    // Parse date from cell
    const dn=target.querySelector('.dn');
    if(dn){
      const d=parseInt(dn.textContent);
      dropEvent(touchDrag.evId,curY,curM,d);
    }
  }
  document.querySelectorAll('.drag-over').forEach(x=>x.classList.remove('drag-over'));
  touchDrag={active:false};
});

function saveEv(){
  const title=document.getElementById('in-title').value.trim();
  if(!title)return;
  const dv=document.getElementById('in-date').value;
  if(!dv)return;
  const type=document.getElementById('in-type').value;
  const gate=parseInt(document.getElementById('in-gate').value);
  const rec=document.getElementById('in-rec').value;
  const desc=document.getElementById('in-desc').value.trim();
  const priority=document.getElementById('in-priority').value;
  const ev={
    id:editId||uid(),
    date:new Date(dv+'T00:00:00'),
    type,gate:gate>=0?gate:-1,
    icon:iconFor(type),
    short:title.slice(0,18),title,
    period:rec==='none'?'One-time':'Repeating '+rec,
    desc,recurring:rec==='none'?false:rec,
    isCustom:true,customColor:selColor,
    priority,daysUntil:dU(dv)
  };
  if(editId){const i=STORE.custom.findIndex(e=>e.id===editId);if(i>=0)STORE.custom[i]=ev;}
  else STORE.custom.push(ev);
  closeM();refresh();renderAll();
  showToast(editId?'Event updated':'Event added');
}

function editC(id){
  const baseId=id.replace(/-\d{4}-\d{2}$/,'');
  const ev=STORE.custom.find(e=>e.id===baseId||e.id===id);
  if(!ev)return;
  editId=ev.id;
  document.getElementById('mtxt').textContent='Edit Event';
  document.getElementById('in-title').value=ev.title;
  document.getElementById('in-date').value=new Date(ev.date).toISOString().slice(0,10);
  document.getElementById('in-type').value=ev.type;
  document.getElementById('in-rec').value=ev.recurring||'none';
  document.getElementById('in-gate').value=ev.gate>=0?ev.gate:'-1';
  document.getElementById('in-priority').value=ev.priority||'normal';
  document.getElementById('in-desc').value=ev.desc||'';
  selColor=ev.customColor||'#E8192C';
  document.querySelectorAll('.cs').forEach(s=>s.classList.toggle('sl',s.dataset.c===selColor));
  document.getElementById('dupBtn').style.display='inline-flex';
  document.getElementById('modal').classList.add('open');
}

function delC(id){
  const baseId=id.replace(/-\d{4}-\d{2}$/,'');
  STORE.custom=STORE.custom.filter(e=>e.id!==baseId&&e.id!==id);
  refresh();renderAll();
  showToast('Event deleted');
}

function dupEv(){
  if(!editId)return;
  const ev=STORE.custom.find(e=>e.id===editId);
  if(!ev)return;
  const nev={...ev,id:uid(),title:ev.title+' (copy)',short:(ev.title+' (copy)').slice(0,18)};
  STORE.custom.push(nev);
  closeM();refresh();renderAll();
  showToast('Event duplicated');
}

function dupFromId(id){
  const baseId=id.replace(/-\d{4}-\d{2}$/,'');
  const ev=STORE.custom.find(e=>e.id===baseId||e.id===id);
  if(!ev){showToast("System events can't be duplicated directly — create a custom event instead.");return;}
  editId=ev.id;dupEv();
}
